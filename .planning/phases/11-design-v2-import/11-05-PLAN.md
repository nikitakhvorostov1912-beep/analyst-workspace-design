### T-11-05-5: Playwright E2E auto-test (15 мин)

NEW `frontend/e2e/design-v2.spec.ts`:

```ts
import { test, expect } from "@playwright/test";

test.describe("Design v2 import", () => {
  test("Header renders with brand mark", async ({ page }) => {
    await page.goto("/");
    await expect(page.locator("text=Аналитик")).toBeVisible();
    await expect(page.locator("text=v1.1.0")).toBeVisible();
  });

  test("Channel selector popover opens with search", async ({ page }) => {
    await page.goto("/");
    await page.locator("button:has-text('Выберите подключение'), button:has-text('Моя 1С')").first().click();
    await expect(page.getByPlaceholder("Поиск подключения")).toBeVisible();
  });

  test("Anon toggle changes state", async ({ page }) => {
    await page.goto("/");
    const btn = page.getByRole("button", { name: /Анон/ });
    const before = await btn.getAttribute("aria-pressed");
    await btn.click();
    const after = await btn.getAttribute("aria-pressed");
    expect(before).not.toBe(after);
  });

  test("Onboarding has 4 steps", async ({ page }) => {
    // если onboarding появится — проверить step indicator
    await page.goto("/");
    const dialog = page.getByRole("dialog");
    if (await dialog.isVisible({ timeout: 2000 }).catch(() => false)) {
      await expect(dialog.locator("text=MCP")).toBeVisible();
      await expect(dialog.locator("text=LLM")).toBeVisible();
      await expect(dialog.locator("text=Обучение")).toBeVisible();
      await expect(dialog.locator("text=Готово")).toBeVisible();
    }
  });
});
```
