```tsx
"use client";

import { BookOpen, Check, Cog, PenLine, Search } from "lucide-react";
import { cn } from "@/lib/utils";

export type StageKind = "analyzing" | "learn" | "tool" | "tool_done" | "finalizing";

export interface Stage {
  kind: StageKind;
  tool?: string;
  doneIn?: number;
}

interface StreamingStagesProps {
  stages: Stage[];
  activeIndex: number;
}

const STAGE_META: Record<StageKind, { Icon: React.ComponentType<{ className?: string }>; label: string; tone: "muted" | "accent" | "success" }> = {
  analyzing:  { Icon: Search,   label: "Анализирую",     tone: "muted" },
  learn:      { Icon: BookOpen, label: "Ищу прошлые ответы", tone: "accent" },
  tool:       { Icon: Cog,      label: "Вызываю",         tone: "accent" },
  tool_done:  { Icon: Check,    label: "Получил данные",  tone: "success" },
  finalizing: { Icon: PenLine,  label: "Формирую ответ",  tone: "accent" },
};

const TONE_BG: Record<string, string> = {
  muted: "bg-transparent text-fg-secondary",
  accent: "bg-accent-08 text-accent",
  success: "bg-[rgba(74,222,128,0.1)] text-success",
};

export function StreamingStages({ stages, activeIndex }: StreamingStagesProps) {
  return (
    <div className="inline-flex flex-wrap items-center gap-1 px-3 py-2 rounded-lg bg-bg-surface border border-bd-emphasis text-xs animate-fade-up">
      {stages.map((s, i) => {
        const meta = STAGE_META[s.kind];
        const isActive = i === activeIndex;
        const isDone = i < activeIndex;
        const Icon = isDone ? Check : meta.Icon;
        const toneClass = isDone ? TONE_BG.success : isActive ? TONE_BG[meta.tone] : "bg-transparent text-fg-subtle";

        return (
          <span key={i} className="inline-flex items-center gap-1">
            {i > 0 && <span className={cn("text-[12px]", i <= activeIndex ? "text-bd-strong" : "text-bd-emphasis")}>→</span>}
            <span
              className={cn(
                "inline-flex items-center gap-1 px-1.5 py-0.5 rounded transition-all duration-normal ease-design-ease",
                toneClass,
                i > activeIndex && "opacity-45"
              )}
            >
              <span className={cn("inline-flex", isActive && s.kind === "tool" && "animate-spin")}>
                <Icon className="h-3 w-3" />
              </span>
              {s.kind === "tool" ? (
                <>
                  <span>{meta.label}</span>
                  <span className="font-mono text-fg-primary">{s.tool}</span>
                  {isActive && <span className="inline-block h-1.5 w-1.5 rounded-full bg-current animate-blink" />}
                </>
              ) : s.kind === "tool_done" ? (
                <span>{meta.label} ({s.doneIn}ms)</span>
              ) : (
                <span>{meta.label}{isActive ? "…" : ""}</span>
              )}
            </span>
          </span>
        );
      })}
    </div>
  );
}
```

### T-11-04-2: `ToolTrace.tsx` рефакторинг (30 мин)

REWRITE `frontend/components/chat/ToolTrace.tsx` под новый дизайн:

```tsx
"use client";

import { useState } from "react";
import { ChevronRight } from "lucide-react";
import { cn } from "@/lib/utils";
import { JsonTree } from "./JsonTree";
import type { ToolCall } from "@/lib/types";

interface ToolTraceProps {
  trace: ToolCall[];
}

export function ToolTrace({ trace }: ToolTraceProps) {
  const [open, setOpen] = useState(false);
  const [expandedIdx, setExpandedIdx] = useState<number | null>(null);
  if (!trace || trace.length === 0) return null;
  const total = trace.reduce((a, t) => a + (t.duration_ms ?? 0), 0);

  return (
    <div className="mt-2">
      <button
        onClick={() => setOpen(!open)}
        className="inline-flex items-center gap-1.5 text-xs text-fg-muted hover:text-fg-primary transition-colors duration-micro ease-design-ease"
      >
        <ChevronRight
          className={cn(
            "h-3.5 w-3.5 transition-transform duration-micro ease-design-ease",
            open && "rotate-90"
          )}
        />
        <span>
          {trace.length} tool {trace.length === 1 ? "call" : "calls"}
        </span>
        <span className="font-mono text-fg-subtle">· {total}ms</span>
        {!open &&
          trace.map((t, i) => (
            <span
              key={i}
              className="font-mono text-[10.5px] px-1.5 py-px bg-bg-raised border border-bd-emphasis rounded text-fg-secondary ml-0.5"
              title={t.name}
            >
              {t.name}
            </span>
          ))}
      </button>
      {open && (
        <div className="mt-2 space-y-1.5 animate-fade-up">
          {trace.map((t, i) => (
            <div key={i} className="bg-bg-surface border border-bd-subtle rounded-md overflow-hidden">
              <button
                onClick={() => setExpandedIdx(expandedIdx === i ? null : i)}
                className="flex items-center gap-2 px-2.5 py-1.5 w-full text-left hover:bg-bg-raised transition-colors duration-micro ease-design-ease"
              >
                <span
                  className={cn(
                    "h-1.5 w-1.5 rounded-full",
                    t.ok ? "bg-success" : "bg-error"
                  )}
                />
                <span className="font-mono text-[12px] text-fg-primary">{t.name}</span>
                <span className="font-mono text-[11px] text-fg-muted ml-auto">{t.duration_ms}ms</span>
              </button>
              {expandedIdx === i && (
                <div className="px-2.5 py-2 border-t border-bd-subtle bg-bg-root text-xs">
                  <div className="text-fg-muted mb-1">args:</div>
                  <JsonTree value={t.arguments} />
                  {t.result && (
                    <>
                      <div className="text-fg-muted mt-2 mb-1">result:</div>
                      <JsonTree value={t.result} />
                    </>
                  )}
                </div>
              )}
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
```

interface CardSkeletonProps {
  rows?: number;

export function CardSkeleton({ rows = 3, className }: CardSkeletonProps) {
  const widths = ["w-full", "w-3/4", "w-5/6", "w-2/3"];
  return (
    <div className={cn("bg-bg-surface border border-bd-subtle rounded-md p-3 animate-pulse min-h-32", className)}>

        <div className="h-7 w-7 rounded-md bg-bg-raised" />
        <div className="flex-1 space-y-1.5">
          <div className="h-3.5 bg-bg-raised rounded w-1/3" />
          <div className="h-2.5 bg-bg-raised rounded w-1/4" />
        </div>
      </div>
      <div className="space-y-2">
        {Array.from({ length: rows }).map((_, i) => (
          <div key={i} className={cn("h-3 bg-bg-raised rounded", widths[i % widths.length])} />
        ))}
      </div>
    </div>
  );
}
