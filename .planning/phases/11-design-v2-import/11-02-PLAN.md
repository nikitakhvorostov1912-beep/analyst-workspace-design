```tsx
"use client";

import { cn } from "@/lib/utils";

export type ConnectionStatus = "online" | "offline" | "connecting";

interface StatusDotProps {
  status: ConnectionStatus;
  size?: "sm" | "md";
  className?: string;
  "aria-label"?: string;
}

const SIZES = {
  sm: "h-1.5 w-1.5",
  md: "h-2 w-2",
};

const COLORS: Record<ConnectionStatus, string> = {
  online: "bg-success animate-status-pulse",
  offline: "bg-error",
  connecting: "bg-warning animate-blink",
};

export function StatusDot({ status, size = "md", className, ...props }: StatusDotProps) {
  return (
    <span
      role="status"
      aria-label={props["aria-label"] ?? `Статус: ${status}`}
      className={cn("inline-block rounded-full", SIZES[size], COLORS[status], className)}
    />
  );
}
```

### T-11-02-2: `CardActionMenu` component (20 мин)

Использовать shadcn `DropdownMenu` вместо самопального popover.

NEW `frontend/components/cards/CardActionMenu.tsx`:

```tsx
"use client";

import { MoreHorizontal } from "lucide-react";
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuSeparator,
  DropdownMenuTrigger,
} from "@/components/ui/dropdown-menu";
import { Button } from "@/components/ui/button";

export interface CardActionItem {
  label: string;
  icon?: React.ComponentType<{ className?: string }>;
  onClick?: () => void;
  kbd?: string;
  destructive?: boolean;
}

interface CardActionMenuProps {
  items: (CardActionItem | "separator")[];
  align?: "start" | "end";
}

export function CardActionMenu({ items, align = "end" }: CardActionMenuProps) {
  return (
    <DropdownMenu>
      <DropdownMenuTrigger asChild>
        <Button variant="ghost" size="icon" className="h-7 w-7" aria-label="Действия">
          <MoreHorizontal className="h-4 w-4" />
        </Button>
      </DropdownMenuTrigger>
      <DropdownMenuContent align={align} className="min-w-[180px] animate-scale-in">
        {items.map((item, i) => {
          if (item === "separator") return <DropdownMenuSeparator key={`sep-${i}`} />;
          const Icon = item.icon;
          return (
            <DropdownMenuItem
              key={item.label}
              onClick={item.onClick}
              className={item.destructive ? "text-error focus:text-error" : ""}
            >
              {Icon && <Icon className="h-3.5 w-3.5 mr-2" />}
              <span className="flex-1">{item.label}</span>
              {item.kbd && (
                <kbd className="ml-auto text-xs text-fg-muted bg-bg-raised px-1.5 py-0.5 rounded">
                  {item.kbd}
                </kbd>
              )}
            </DropdownMenuItem>
          );
        })}
      </DropdownMenuContent>
    </DropdownMenu>
  );
}
```

```tsx
"use client";

import { Table, FileText, FileClock, TrendingUp, Link as LinkIcon, Code2, Lock, Unlock } from "lucide-react";
import { cn } from "@/lib/utils";
import { CardActionMenu, type CardActionItem } from "./CardActionMenu";

export type CardType = "table" | "object" | "log" | "metric" | "references" | "code";

const TYPE_META: Record<CardType, { icon: React.ComponentType<{ className?: string }>; label: string; accentClass: string }> = {
  table:      { icon: Table,       label: "Таблица",  accentClass: "text-accent" },
  object:     { icon: FileText,    label: "Объект",   accentClass: "text-accent" },
  log:        { icon: FileClock,   label: "Журнал",   accentClass: "text-warning" },
  metric:     { icon: TrendingUp,  label: "Метрика",  accentClass: "text-success" },
  references: { icon: LinkIcon,    label: "Ссылки",   accentClass: "text-accent" },
  code:       { icon: Code2,       label: "Код",      accentClass: "text-accent" },
};

interface CardHeaderProps {
  type: CardType;
  title: string;
  meta?: string;
  toolName?: string;
  anonymizable?: boolean;
  anonOn?: boolean;
  onAnonToggle?: () => void;
  actions?: (CardActionItem | "separator")[];
}

export function CardHeader({ type, title, meta, toolName, anonymizable, anonOn, onAnonToggle, actions = [] }: CardHeaderProps) {
  const meta_ = TYPE_META[type];
  const TypeIcon = meta_.icon;

  return (
    <div className="flex items-start gap-2.5 px-3 py-3 border-b border-bd-subtle">
      <div
        className={cn(
          "h-7 w-7 rounded-md inline-flex items-center justify-center bg-bg-raised border border-bd-emphasis flex-shrink-0",
          meta_.accentClass
        )}
      >
        <TypeIcon className="h-3.5 w-3.5" />
      </div>
      <div className="min-w-0 flex-1">
        <div className="font-medium text-fg-primary text-[13.5px] leading-tight">{title}</div>
        <div className="flex items-center gap-2 mt-1 text-xs text-fg-muted flex-wrap">
          {toolName && (
            <span className="font-mono px-1.5 py-px bg-bg-raised border border-bd-emphasis rounded text-[11px] text-fg-secondary">
              {toolName}
            </span>
          )}
          {meta && <span>{meta}</span>}
        </div>
      </div>
      <div className="flex items-center gap-1">
        {anonymizable && (
          <button
            onClick={onAnonToggle}
            className={cn(
              "inline-flex items-center gap-1.5 px-2 py-1 rounded text-[11.5px] border transition-colors duration-micro ease-design-ease",
              anonOn
                ? "bg-warning-12 text-warning border-warning-20"
                : "text-fg-muted border-bd-emphasis"
            )}
            title={anonOn ? "Раскрыть значения" : "Включить маски"}
          >
            {anonOn ? <Lock className="h-3 w-3" /> : <Unlock className="h-3 w-3" />}
            <span>{anonOn ? "Маскировано" : "Раскрыто"}</span>
          </button>
        )}
        {actions.length > 0 && <CardActionMenu items={actions} />}
      </div>
    </div>
  );
}
```

Test ≥ 4 шт.: rendering каждого type, anon toggle, action menu render, meta + toolName display.

import { Button } from "@/components/ui/button";
import { cn } from "@/lib/utils";

interface EmptyStateProps {
  icon?: React.ComponentType<{ className?: string }>;
  title: string;
  description?: string;
  cta?: {
    label: string;
    onClick?: () => void;
    href?: string;
  };
  className?: string;

export function EmptyState({ icon: Icon, title, description, cta, className }: EmptyStateProps) {
  return (
    <div className={cn("flex flex-col items-center justify-center text-center max-w-md mx-auto py-12 px-6", className)}>
      {Icon && (
        <div className="h-12 w-12 rounded-full bg-accent-08 border border-accent-20 text-accent inline-flex items-center justify-center mb-4">
          <Icon className="h-6 w-6" />
        </div>
      )}
      <h3 className="text-base font-medium text-fg-primary mb-1.5">{title}</h3>
      {description && <p className="text-sm text-fg-muted leading-relaxed">{description}</p>}
      {cta && (
        <div className="mt-5">
          {cta.href ? (
            <a href={cta.href}>
              <Button>{cta.label}</Button>
            </a>
          ) : (
            <Button onClick={cta.onClick}>{cta.label}</Button>
          )}
        </div>
      )}
    </div>
  );
}

### T-11-02-5: `ErrorBanner` component (15 мин)

NEW `frontend/components/ui/ErrorBanner.tsx`:

```tsx
import { AlertCircle, AlertTriangle, Info, X, RotateCw } from "lucide-react";
import { Button } from "@/components/ui/button";
import { cn } from "@/lib/utils";

export type ErrorSeverity = "info" | "warning" | "error";

const SEVERITY: Record<ErrorSeverity, { icon: React.ComponentType<{ className?: string }>; classes: string }> = {
  info: {
    icon: Info,
    classes: "bg-accent-08 border-accent-20 text-accent",
  },
  warning: {
    icon: AlertTriangle,
    classes: "bg-warning-12 border-warning-20 text-warning",
  },
  error: {
    icon: AlertCircle,
    classes: "bg-[rgba(248,113,113,0.08)] border-[rgba(248,113,113,0.2)] text-error",
  },
};

interface ErrorBannerProps {
  severity?: ErrorSeverity;
  title: string;
  description?: string;
  onRetry?: () => void;
  onDismiss?: () => void;
  className?: string;
}

export function ErrorBanner({ severity = "error", title, description, onRetry, onDismiss, className }: ErrorBannerProps) {
  const s = SEVERITY[severity];
  const Icon = s.icon;
  return (
    <div
      role="alert"
      className={cn(
        "flex items-start gap-3 p-3 rounded-md border animate-fade-up",
        s.classes,
        className
      )}
    >
      <Icon className="h-4 w-4 flex-shrink-0 mt-0.5" />
      <div className="flex-1 min-w-0">
        <div className="text-sm font-medium text-fg-primary">{title}</div>
        {description && <div className="text-xs text-fg-muted mt-1">{description}</div>}
      </div>
      <div className="flex items-center gap-1">
        {onRetry && (
          <Button variant="ghost" size="sm" onClick={onRetry} className="h-7 px-2">
            <RotateCw className="h-3 w-3 mr-1.5" />
            Повторить
          </Button>
        )}
        {onDismiss && (
          <Button variant="ghost" size="icon" onClick={onDismiss} className="h-7 w-7" aria-label="Закрыть">
            <X className="h-3.5 w-3.5" />
          </Button>
        )}
      </div>
    </div>
  );
}
```

Test ≥ 4: каждой severity + retry click + dismiss click.
