### T-11-03-2: `ChannelSelector.tsx` рефакторинг с shadcn Popover (40 мин)

REWRITE `frontend/components/shell/ChannelSelector.tsx`:

```tsx
"use client";

import { useEffect, useState } from "react";
import { Check, ChevronDown, Plus, Search, Settings } from "lucide-react";
import { Popover, PopoverContent, PopoverTrigger } from "@/components/ui/popover";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { StatusDot, type ConnectionStatus } from "@/components/ui/StatusDot";
import { fetchConnections } from "@/lib/api";
import type { Connection } from "@/lib/types";

interface ChannelSelectorProps {
  onOpenSettings?: () => void;
}

export function ChannelSelector({ onOpenSettings }: ChannelSelectorProps) {
  const [open, setOpen] = useState(false);
  const [search, setSearch] = useState("");
  const [connections, setConnections] = useState<Connection[]>([]);
  const [activeId, setActiveId] = useState<string | null>(null);

  useEffect(() => {
    fetchConnections().then((c) => {
      setConnections(c);
      const stored = typeof window !== "undefined" ? localStorage.getItem("activeChannelId") : null;
      setActiveId(stored ?? c[0]?.id ?? null);
    });
  }, []);

  const active = connections.find((c) => c.id === activeId);
  const filtered = search.trim()
    ? connections.filter((c) => c.name.toLowerCase().includes(search.toLowerCase()))
    : connections;

  function handleSelect(id: string) {
    setActiveId(id);
    localStorage.setItem("activeChannelId", id);
    setOpen(false);
  }

  return (
    <Popover open={open} onOpenChange={setOpen}>
      <PopoverTrigger asChild>
        <button className="inline-flex items-center gap-2 px-3 h-8 bg-bg-surface border border-bd-emphasis rounded-md max-w-[480px] hover:border-bd-strong transition-colors duration-micro ease-design-ease">
          <StatusDot status={(active?.status as ConnectionStatus) ?? "offline"} size="sm" />
          <span className="font-medium text-[13px] text-fg-primary truncate">
            {active?.name ?? "Выберите подключение"}
          </span>
          {active?.endpoint && (
            <span className="font-mono text-[11.5px] text-fg-muted truncate">{active.endpoint}</span>
          )}
          <ChevronDown className="h-3.5 w-3.5 text-fg-subtle flex-shrink-0" />
        </button>
      </PopoverTrigger>
      <PopoverContent className="w-[420px] p-1 animate-scale-in" align="center">
        <div className="flex items-center gap-1.5 px-2.5 py-2 border-b border-bd-subtle">
          <Search className="h-3.5 w-3.5 text-fg-muted flex-shrink-0" />
          <Input
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            placeholder="Поиск подключения..."
            className="border-0 bg-transparent h-7 px-0 text-sm focus-visible:ring-0"
          />
        </div>
        <div className="max-h-[320px] overflow-auto py-1">
          {filtered.length === 0 ? (
            <div className="px-3 py-4 text-center text-xs text-fg-muted">Ничего не найдено</div>
          ) : (
            filtered.map((c) => (
              <button
                key={c.id}
                onClick={() => handleSelect(c.id)}
                className={`flex items-center gap-2.5 px-2.5 py-2 w-full rounded text-left transition-colors duration-micro ease-design-ease ${
                  c.id === activeId ? "bg-accent-08" : "hover:bg-bg-raised"
                }`}
              >
                <StatusDot status={(c.status as ConnectionStatus) ?? "offline"} size="sm" />
                <div className="flex-1 min-w-0">
                  <div className="text-[13px] font-medium text-fg-primary truncate">{c.name}</div>
                  <div className="text-[11px] text-fg-muted font-mono truncate">
                    {c.endpoint}
                    {c.lastPing && ` · ${c.lastPing}`}
                  </div>
                </div>
                {c.id === activeId && <Check className="h-3.5 w-3.5 text-accent flex-shrink-0" />}
              </button>
            ))
          )}
        </div>
        <div className="flex gap-1.5 px-1.5 pt-1.5 pb-1 border-t border-bd-subtle mt-1">
          <Button variant="ghost" size="sm" onClick={() => { setOpen(false); onOpenSettings?.(); }} className="h-7 text-xs">
            <Plus className="h-3 w-3 mr-1" /> Добавить
          </Button>
          <Button variant="ghost" size="sm" onClick={() => { setOpen(false); onOpenSettings?.(); }} className="h-7 text-xs">
            <Settings className="h-3 w-3 mr-1" /> Настроить
          </Button>
        </div>
      </PopoverContent>
    </Popover>
  );
}
```

### T-11-03-4: `ModelBadge.tsx` рефакторинг (10 мин)

REWRITE:

```tsx
"use client";

import { useEffect, useState } from "react";
import { Sparkles } from "lucide-react";
import { fetchLLMConfig } from "@/lib/api";

export function ModelBadge() {
  const [model, setModel] = useState("");
  const [temp, setTemp] = useState<number | null>(null);

  useEffect(() => {
    fetchLLMConfig().then((c) => {
      setModel(c?.model ?? "—");
      setTemp(c?.temperature ?? null);
    });
  }, []);

  if (!model) return null;
  return (
    <div className="inline-flex items-center gap-1.5 px-2.5 h-[30px] rounded-md bg-bg-surface border border-bd-emphasis text-xs text-fg-secondary">
      <Sparkles className="h-3 w-3 text-accent" />
      <span className="font-mono">{model}</span>
      {temp !== null && <span className="font-mono text-fg-subtle">· {temp.toFixed(1)}</span>}
    </div>
  );
}
```
