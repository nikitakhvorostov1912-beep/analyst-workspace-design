### T-09-01-3: Backend admin/reset endpoint (30 мин)

NEW `backend/app/routes/admin.py`:

```python
"""Admin endpoints (privacy escape hatches)."""
from fastapi import APIRouter, Depends, Header, HTTPException, status
from app.storage.db import DB, get_db
import structlog

logger = structlog.get_logger()
router = APIRouter(prefix="/admin", tags=["admin"])


@router.post("/reset-local-db", status_code=status.HTTP_200_OK)
async def reset_local_db(
    x_confirm_reset: str = Header(..., alias="X-Confirm-Reset"),
    db: DB = Depends(get_db),
) -> dict:
    """Truncate session/message data. Preserve settings (LLM, MCP connections).

    Requires explicit X-Confirm-Reset: true header to prevent accidental calls.
    """
    if x_confirm_reset != "true":
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Missing or invalid X-Confirm-Reset header (must be 'true')",
        )

    cleared = []
    for table in (
        "messages",
        "sessions",
        "tool_calls",
        "anon_tokens",
        "metadata_cache",
        "card_states",
    ):
        try:
            await db.execute(f"DELETE FROM {table}")
            cleared.append(table)
        except Exception as e:
            logger.warning(f"reset_table_failed", table=table, error=str(e))

    await db.commit()
    logger.info("local_db_reset", cleared=cleared)
    return {"status": "ok", "cleared": cleared}
```

### T-09-01-4: Frontend api.ts + LocalDataSection component (30 мин)

EDIT `frontend/lib/api.ts` — добавить:

```ts
export async function resetLocalDb(): Promise<{ ok: boolean; cleared?: string[] }> {
  const r = await fetch(`${BACKEND_URL}/admin/reset-local-db`, {
    method: "POST",
    headers: { "X-Confirm-Reset": "true" },
  });
  if (!r.ok) return { ok: false };
  const data = await r.json();
  return { ok: true, cleared: data.cleared };
}
```

NEW `frontend/components/settings/LocalDataSection.tsx`:

```tsx
"use client";

import { useState } from "react";
import { Button } from "@/components/ui/button";
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
  AlertDialogTrigger,
} from "@/components/ui/alert-dialog";
import { resetLocalDb } from "@/lib/api";
import { toast } from "sonner";

export function LocalDataSection() {
  const [loading, setLoading] = useState(false);

  async function handleReset() {
    setLoading(true);
    const r = await resetLocalDb();
    setLoading(false);
    if (r.ok) {
      toast.success(`Локальная база сброшена (${r.cleared?.join(", ") ?? ""})`);
      setTimeout(() => window.location.reload(), 1000);
    } else {
      toast.error("Не удалось сбросить базу");
    }
  }

  return (
    <section className="space-y-3 border-t pt-6">
      <h3 className="text-lg font-medium">Локальные данные</h3>
      <p className="text-sm text-muted-foreground">
        Удаляет все сессии чата, токены анонимизации, кеш метаданных и состояния карточек.
        Настройки MCP и LLM сохраняются.
      </p>
      <AlertDialog>
        <AlertDialogTrigger asChild>
          <Button variant="destructive" disabled={loading}>
            {loading ? "Сбрасываю..." : "Сбросить локальную базу"}
          </Button>
        </AlertDialogTrigger>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Точно сбросить базу?</AlertDialogTitle>
            <AlertDialogDescription>
              Действие необратимо. Все ваши сессии чата будут удалены.
              Настройки подключений останутся.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>Отмена</AlertDialogCancel>
            <AlertDialogAction onClick={handleReset}>Сбросить</AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </section>
  );
}
```

## Risks

- **Кириллица в `1С Аналитик`** — путь `%APPDATA%\1С Аналитик\` non-ASCII. SQLite + Windows + Python ОК если URL с forward-slashes. **Mitigation:** explicit test в smoke step 4
- **PyInstaller --onefile %TEMP%** — backend.exe распаковывается в `%TEMP%\_MEI*` при каждом запуске. **Это runtime extract, не data.** Главное чтобы DATABASE_URL показывал на APPDATA, а не на TEMP — verify в console output
- **NSIS uninstaller удалит APPDATA** — теоретически NSIS `RMDir /r` мог бы зацепить APPDATA. **Mitigation:** в текущем electron-builder.yml `oneClick: false, perMachine: false` — по умолчанию appData НЕ удаляется. Verify в smoke step 9
- **Permissions** — `%APPDATA%\1С Аналитик\` создаётся при первом запуске. Если права read-only — приложение не запустится. **Mitigation:** в `desktop/main.js` добавить `try/catch` с понятной ошибкой
