```python
class ForgetSessionRequest(BaseModel):
    session_id: int


@router.post("/forget-session")
async def forget_session(
    body: ForgetSessionRequest,
    db: DB = Depends(get_db),
):
    """Remove embeddings for all messages in a single session."""
    rows = await db.execute(
        "SELECT id FROM messages WHERE session_id = ?",
        (body.session_id,),
    )
    rows = await rows.fetchall()
    msg_ids = [r["id"] for r in rows]
    if not msg_ids:
        return {"deleted": 0}
    placeholders = ",".join("?" * len(msg_ids))
    await db.execute(
        f"DELETE FROM message_embeddings WHERE message_id IN ({placeholders})",
        tuple(msg_ids),
    )
    await db.commit()
    logger.info("forget_session", session_id=body.session_id, deleted=len(msg_ids))
    return {"deleted": len(msg_ids)}


@router.post("/forget-all")
async def forget_all(
    x_confirm_forget: str = Header(..., alias="X-Confirm-Forget"),
    db: DB = Depends(get_db),
):
    """Remove ALL embeddings (full reset of learn data)."""
    if x_confirm_forget != "true":
        raise HTTPException(400, "Missing X-Confirm-Forget: true")
    r = await db.execute("DELETE FROM message_embeddings")
    await db.commit()
    return {"deleted": r.rowcount}


@router.get("/stats")
async def learn_stats(db: DB = Depends(get_db)):
    """Aggregate stats for Settings UI."""
    total_msgs = await (await db.execute("SELECT COUNT(*) AS c FROM messages")).fetchone()
    indexed = await (await db.execute("SELECT COUNT(*) AS c FROM message_embeddings")).fetchone()
    settings = await (await db.execute("SELECT learn_enabled FROM llm_settings WHERE id = 1")).fetchone()
    return {
        "learn_enabled": bool(settings.get("learn_enabled") if settings else False),
        "total_messages": total_msgs["c"],
        "indexed_messages": indexed["c"],
        "coverage_pct": round(indexed["c"] / max(total_msgs["c"], 1) * 100, 1),
    }
```

```tsx
  AlertDialogFooter, AlertDialogAction, AlertDialogCancel,

import { toast } from "sonner";
import { BookOpen, Trash2 } from "lucide-react";
import { fetchLearnStats, forgetAll, setLearnEnabled } from "@/lib/api";

export function LearnSection() {
  const [enabled, setEnabled] = useState(false);
  const [stats, setStats] = useState<{
    total: number; indexed: number; coverage: number;
  } | null>(null);

  useEffect(() => {
    (async () => {
      const s = await fetchLearnStats();
      setEnabled(s.learn_enabled);
      setStats({ total: s.total_messages, indexed: s.indexed_messages, coverage: s.coverage_pct });
    })();
  }, []);

  async function handleToggle(checked: boolean) {
    const r = await setLearnEnabled(checked);
    if (r.ok) {
      setEnabled(checked);
      toast.success(checked ? "Обучение включено" : "Обучение выключено");

    const r = await forgetAll();
    if (r.ok) {
      toast.success(`Удалено ${r.deleted} embeddings`);
      const s = await fetchLearnStats();
      setStats({ total: s.total_messages, indexed: s.indexed_messages, coverage: s.coverage_pct });
    }
  }

  return (
    <section className="space-y-4 border-t pt-6">
      <header className="flex items-center gap-2">
        <BookOpen className="h-5 w-5" />
        <h3 className="text-lg font-medium">Обучение на ваших сессиях</h3>
      </header>

      <p className="text-sm text-muted-foreground">
        Когда включено, LLM будет использовать ваши прошлые сессии как контекст
        для текущих ответов. Данные не покидают ваш компьютер кроме embeddings
        (короткое числовое представление текста) — они отправляются в настроенный
        вами Embeddings API.
      </p>

      <div className="flex items-center justify-between">
        <Label htmlFor="learn-toggle">Включить обучение</Label>
        <Switch id="learn-toggle" checked={enabled} onCheckedChange={handleToggle} />
      </div>

      {stats && (
        <div className="text-sm text-muted-foreground space-y-1 bg-secondary p-3 rounded-md">
          <div>Всего сообщений: <span className="font-mono">{stats.total}</span></div>
          <div>Проиндексировано: <span className="font-mono">{stats.indexed}</span> ({stats.coverage}%)</div>
        </div>
      )}

      <AlertDialog>
        <AlertDialogTrigger asChild>
          <Button variant="destructive" size="sm">
            <Trash2 className="h-4 w-4 mr-2" />
```

```ts
export async function fetchLearnStats() {
  const r = await fetch(`${BACKEND_URL}/learn/stats`);
  return await r.json();
}

export async function setLearnEnabled(enabled: boolean) {
  const r = await fetch(`${BACKEND_URL}/llm-config`, {
    method: "PATCH",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ learn_enabled: enabled }),
  });
  return { ok: r.ok };
}

export async function forgetSession(sessionId: number) {
  const r = await fetch(`${BACKEND_URL}/learn/forget-session`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ session_id: sessionId }),
  });
  return await r.json();
}

export async function forgetAll() {
  const r = await fetch(`${BACKEND_URL}/learn/forget-all`, {
    method: "POST",
    headers: { "X-Confirm-Forget": "true" },
  });
  if (!r.ok) return { ok: false };
  return { ok: true, deleted: (await r.json()).deleted };
}
```

```markdown
## Обучение на ваших сессиях (LEARN, опционально)

Приложение может использовать ваши прошлые сессии как контекст для текущих
ответов LLM. Например, если вы вчера спрашивали про поручительства, а сегодня
вернулись к той же теме — LLM найдёт прошлый ответ и сошлётся на него.

### Как это работает

1. Все сообщения индексируются в локальной vector-БД (SQLite-vec в `%APPDATA%\1С Аналитик\app.db`)
2. Перед каждым ответом LLM ищется 3 самых похожих сообщения из прошлых сессий
3. Если найдены релевантные (similarity ≥ 70%) — встраиваются в system prompt
4. Под ответом показывается badge «📚 N сообщений из прошлых сессий»

### Что отправляется в облако

- Тексты ваших сообщений → embeddings API (OpenAI / Yandex / self-hosted BGE-M3)
- В ответ приходит вектор (числа), который хранится локально
- LLM получает только короткие snippets из прошлых сессий (≤ 300 символов × 3 шт.)

### Privacy controls

- **Default OFF** — обучение выключено пока вы не включите его сами
- **Settings → Обучение** — toggle ON/OFF в любой момент
- **Удалить всё проиндексированное** — кнопка в Settings
- **Удалить конкретную сессию** — через API (`POST /learn/forget-session`)
- **Сбросить локальную базу** (Phase 9) — удаляет всё, включая embeddings
- **Отдельная настройка Embeddings API** — можете использовать self-hosted модель
  (BGE-M3 в Docker) если не хотите отправлять данные в облако
```

```markdown
### Q: Что такое «📚 N сообщений» под ответом?

A: LLM нашла N сообщений из ваших прошлых сессий, которые похожи на ваш текущий
вопрос, и использовала их как контекст. Клик по badge → откроется список источников.

A: Settings → Обучение на ваших сессиях → toggle OFF.
```

## Risks

- **Onboarding wizard размер** — добавление 4-го шага может усложнить first-run. **Mitigation:** Learn step explicitly помечен как «опционально», skip button работает
- **Stats endpoint performance** — `COUNT(*) FROM message_embeddings` может быть медленным на 100K+ записях. **Mitigation:** SQLite-vec автоматически индексирует; verify через EXPLAIN
- **Forget-session гонка с background job** — если job индексирует в момент forget. **Mitigation:** обе операции в transactions; idempotent
