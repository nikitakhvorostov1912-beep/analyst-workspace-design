### T-10-01-2: Embeddings service (45 мин)

NEW `backend/app/services/embeddings.py`:

```python
"""OpenAI-compatible embeddings service."""
from __future__ import annotations

import httpx
import structlog
from typing import Optional

logger = structlog.get_logger()


class EmbeddingsService:
    """OpenAI-compatible embeddings client.

    Works with OpenAI, Yandex (via shim), self-hosted BGE-M3, etc.
    """

    def __init__(
        self,
        endpoint: str,
        model: str,
        api_key: str,
        dim: int = 1536,
        timeout: float = 30.0,
    ):
        self.endpoint = endpoint.rstrip("/")
        self.model = model
        self.api_key = api_key
        self.dim = dim
        self._client = httpx.AsyncClient(timeout=timeout, trust_env=False)

    async def embed(self, text: str) -> list[float]:
        r = await self._client.post(
            f"{self.endpoint}/embeddings",
            json={"model": self.model, "input": text[:8000]},
            headers={"Authorization": f"Bearer {self.api_key}"},
        )
        r.raise_for_status()
        data = r.json()["data"][0]["embedding"]
        if len(data) != self.dim:
            logger.warning("embedding_dim_mismatch", expected=self.dim, got=len(data))
        return data

    async def embed_batch(
        self,
        texts: list[str],
        batch_size: int = 100,
    ) -> list[list[float]]:
        result = []
        for i in range(0, len(texts), batch_size):
            batch = [t[:8000] for t in texts[i : i + batch_size]]
            r = await self._client.post(
                f"{self.endpoint}/embeddings",
                json={"model": self.model, "input": batch},
                headers={"Authorization": f"Bearer {self.api_key}"},
            )
            r.raise_for_status()
            for d in r.json()["data"]:
                result.append(d["embedding"])
        return result

    async def close(self):
        await self._client.aclose()


def build_from_settings(settings: dict) -> Optional[EmbeddingsService]:
    """Build service from llm_settings DB row. Returns None if disabled or missing config."""
    if not settings.get("learn_enabled"):
        return None
    endpoint = settings.get("embeddings_endpoint")
    model = settings.get("embeddings_model")
    api_key = settings.get("embeddings_api_key")
    dim = settings.get("embeddings_dim", 1536)
    if not (endpoint and model and api_key):
        return None
    return EmbeddingsService(endpoint=endpoint, model=model, api_key=api_key, dim=dim)
```

### T-10-01-3: Background indexing job (45 мин)

NEW `backend/app/jobs/index_messages.py`:

```python
"""Background job: index pending messages into vector store."""
from __future__ import annotations

import asyncio
import json
import structlog
from app.storage.db import get_db
from app.services.embeddings import build_from_settings

logger = structlog.get_logger()


async def get_llm_settings_dict(db) -> dict:
    row = await db.execute("SELECT * FROM llm_settings WHERE id = 1")
    r = await row.fetchone()
    return dict(r) if r else {}


async def index_pending(db, emb_service, limit: int = 100) -> int:
    """Index up to `limit` unprocessed messages. Returns count indexed."""
    rows = await db.execute(
        """
        SELECT m.id, m.content
        FROM messages m
        LEFT JOIN message_embeddings me ON me.message_id = m.id
        WHERE me.message_id IS NULL
          AND m.role IN ('user', 'assistant')
          AND m.content IS NOT NULL
          AND LENGTH(m.content) >= 10
        ORDER BY m.id ASC
        LIMIT ?
        """,
        (limit,),
    )
    rows = await rows.fetchall()
    if not rows:
        return 0
    texts = [r["content"] for r in rows]
    embeddings = await emb_service.embed_batch(texts)
    for row, vec in zip(rows, embeddings):
        await db.execute(
            "INSERT INTO message_embeddings(message_id, embedding) VALUES (?, ?)",
            (row["id"], json.dumps(vec)),
        )
    await db.commit()
    return len(rows)


async def run_indexing_loop(interval_sec: int = 60):
    """Forever loop: index every interval_sec, respect learn_enabled toggle."""
    while True:
        try:
            db = await get_db()
            settings = await get_llm_settings_dict(db)
            if not settings.get("learn_enabled"):
                await asyncio.sleep(interval_sec)
                continue
            emb = build_from_settings(settings)
            if emb is None:
                logger.warning("embeddings_not_configured")
                await asyncio.sleep(interval_sec)
                continue
            n = await index_pending(db, emb)
            if n:
                logger.info("indexed_messages", count=n)
            await emb.close()
        except Exception as e:
            logger.error("indexing_loop_failed", error=str(e))
        await asyncio.sleep(interval_sec)
```

## Risks

- **sqlite-vec + PyInstaller** — native DLL. PyInstaller через `--collect-all sqlite_vec` или явный `--add-binary`. **Mitigation:** в Phase 9 уже test PyInstaller bundle — добавим verify для vec0
- **Embeddings API ключ** — может быть невалидным. Background job не должен крашить — try/except + лог warning
- **Стоимость** — индексация 1000 сообщений × ~200 токенов = $0.004 на OpenAI text-embedding-3-small. Копейки, но **показать предупреждение в onboarding Step 3**: «Будет проиндексировано N сообщений (~$X)»
- **Размер embedding column** — 1536 floats × 4 bytes = 6 KB на сообщение. 10000 сообщений = 60 MB. OK для SQLite
- **Race condition** — два инстанса background job (теоретически). **Mitigation:** index_pending использует `LEFT JOIN` для idempotency
