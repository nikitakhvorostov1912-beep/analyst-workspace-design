```python
"""LEARN engine endpoints — retrieve relevant snippets from past sessions."""
from __future__ import annotations

import json
import structlog
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, Field
from app.storage.db import DB, get_db
from app.services.embeddings import build_from_settings

logger = structlog.get_logger()
router = APIRouter(prefix="/learn", tags=["learn"])


class RetrieveRequest(BaseModel):
    query: str = Field(..., min_length=3, max_length=2000)
    channel_id: int | None = None
    top_k: int = Field(default=5, ge=1, le=20)
    threshold: float = Field(default=0.7, ge=0.0, le=1.0)


class Snippet(BaseModel):
    message_id: int
    session_id: int
    session_title: str
    snippet: str
    similarity: float
    created_at: str


class RetrieveResponse(BaseModel):
    snippets: list[Snippet]


@router.post("/retrieve", response_model=RetrieveResponse)
async def retrieve_relevant(
    body: RetrieveRequest,
    db: DB = Depends(get_db),
) -> RetrieveResponse:
    # Check learn_enabled
    settings_row = await (await db.execute("SELECT * FROM llm_settings WHERE id = 1")).fetchone()
    if not settings_row or not settings_row.get("learn_enabled"):
        return RetrieveResponse(snippets=[])

    emb = build_from_settings(dict(settings_row))
    if emb is None:
        logger.warning("retrieve_no_embeddings_service")
        return RetrieveResponse(snippets=[])

    try:
        query_vec = await emb.embed(body.query)
    finally:
        await emb.close()

    # SQLite-vec cosine distance: smaller = more similar
    # similarity = 1 - distance (for cosine)
    distance_threshold = 1.0 - body.threshold

    channel_filter = ""
    params = [json.dumps(query_vec), body.top_k]
    if body.channel_id is not None:
        channel_filter = "AND s.channel_id = ?"
        params.insert(1, body.channel_id)

    rows = await db.execute(
        f"""
        SELECT
            m.id AS message_id,
            m.session_id,
            m.content,
            m.created_at,
            s.title AS session_title,
            vec_distance_cosine(me.embedding, ?) AS dist
        FROM message_embeddings me
        JOIN messages m ON m.id = me.message_id
        JOIN sessions s ON s.id = m.session_id
        WHERE m.role = 'assistant'
          {channel_filter}
        ORDER BY dist ASC
        LIMIT ?
        """,
        tuple(params),
    )
    rows = await rows.fetchall()

    snippets = []
    for r in rows:
        if r["dist"] > distance_threshold:
            continue
        snippets.append(
            Snippet(
                message_id=r["message_id"],
                session_id=r["session_id"],
                session_title=r["session_title"] or "Без названия",
                snippet=r["content"][:500],
                similarity=round(1.0 - r["dist"], 3),
                created_at=r["created_at"],
            )
        )

    return RetrieveResponse(snippets=snippets)
```

async def chat(session_id, message_text, channel_id, db, ...) -> AsyncIterator[SSEEvent]:
    # ... existing: load history, tools, etc.

    # LEARN context retrieval
    learn_context = ""
    learn_snippets = []
    try:
        settings_row = await (await db.execute("SELECT * FROM llm_settings WHERE id = 1")).fetchone()
        if settings_row and settings_row.get("learn_enabled"):
            from app.routes.learn import RetrieveRequest, retrieve_relevant
            from app.services.embeddings import build_from_settings
            emb = build_from_settings(dict(settings_row))
            if emb is not None:
                try:
                    query_vec = await emb.embed(message_text)
                finally:
                    await emb.close()

                # Inline retrieve to avoid HTTP roundtrip
                rows = await db.execute(
                    """SELECT m.id, m.session_id, m.content, m.created_at, s.title,
                              vec_distance_cosine(me.embedding, ?) AS dist
                       FROM message_embeddings me
                       JOIN messages m ON m.id = me.message_id
                       JOIN sessions s ON s.id = m.session_id
                       WHERE m.role = 'assistant' AND s.channel_id = ? AND m.session_id != ?
                       ORDER BY dist ASC LIMIT 5""",
                    (json.dumps(query_vec), channel_id, session_id),
                )
                rows = await rows.fetchall()
                learn_snippets = [
                    {
                        "message_id": r["id"],
                        "session_id": r["session_id"],
                        "session_title": r["title"] or "Без названия",
                        "snippet": r["content"][:500],
                        "similarity": round(1.0 - r["dist"], 3),
                        "created_at": r["created_at"],
                    }
                    for r in rows[:3] if r["dist"] < 0.3  # similarity >= 0.7
                ]
                if learn_snippets:
                    learn_context = (
                        "\n\nНа основе ваших прошлых сессий по этой базе:\n"
                        + "\n\n".join(
                            f"— [{s['session_title']}, {s['created_at'][:10]}]\n  {s['snippet'][:300]}"
                            for s in learn_snippets
                        )
                    )
                    yield SSEEvent(event="learn_context", data=learn_snippets)
    except Exception as e:
        logger.warning("learn_retrieve_failed_skip", error=str(e))

    # Build system prompt with learn_context
    system_prompt = build_system_prompt(tools_schema) + learn_context

    # ... existing LLM call

**Limits:**
- Max 3 snippets included
- Each snippet ≤ 300 chars → ~3 × 300 × 1.3 (overhead) ≈ 1200 tokens
- Exclude current session (`m.session_id != ?`)
- Filter by channel (`s.channel_id = ?`) — не смешиваем разных клиентов

## Acceptance criteria

1. ✅ `POST /learn/retrieve` возвращает top-K snippets с similarity ≥ threshold
2. ✅ Endpoint respects `learn_enabled` toggle (empty array if disabled)
3. ✅ Endpoint фильтрует по channel_id если передан
4. ✅ Orchestrator перед LLM-вызовом делает retrieve + добавляет в system prompt (max 3 snippet × 300 chars)
5. ✅ SSE event `learn_context` приходит до first delta
6. ✅ Frontend `LearnContextBadge` показывается под message с popover
7. ✅ Badge clickable: показывает session_title + snippet preview + ссылку на /sessions/{id}
8. ✅ E2E Playwright: badge появляется во 2-й связанной сессии
9. ✅ Unit tests retrieve ≥ 3
10. ✅ Tokens в system prompt не превышают 1500 (включая learn_context)

## Risks

- **HTTP roundtrip overhead** — если orchestrator вызывает `/learn/retrieve` через HTTP, дополнительные 20-50 мс. **Mitigation:** inline retrieve (как показано в коде) — без HTTP, напрямую к DB
- **Embedding API latency** — embed(query) занимает 100-300 мс перед LLM. **Mitigation:** показать "📚 Ищу прошлые ответы..." как промежуточный SSE status
- **Token limit overflow** — если LLM context window мал (4K), learn_context может выдавить tools schema. **Mitigation:** считать tokens перед добавлением, обрезать learn_context до 1000 tokens hard limit
- **Cross-session leakage** — могут попадать snippets с другими МКП-channels. **Mitigation:** обязательный фильтр `s.channel_id = ?` в SQL
- **Empty channel_id** — если в orchestrator chan_id == None → нет фильтрации → leak между всеми channels. **Mitigation:** assert channel_id is not None в orchestrator
